% Resume template — Stanford Career Education formatting.
%
% Follows Stanford's published guidance rather than the denser Charter pattern
% this replaced:
%   - Margins 0.75in (Stanford: minimum 0.70). The previous 0.5in read cramped,
%     and white space is what lets a recruiter scan.
%   - Helvetica/Arial at 11pt (Stanford: Calibri or Arial, 11-12pt). Carlito,
%     the metric-compatible Calibri clone, is not installed here; TeX Gyre
%     Heros is a Helvetica clone and the closest available match to Arial.
%   - Left-justified throughout: US recruiters scan left to right.
%   - Company bolded (Stanford: bold whichever of company or title lands
%     harder; for this candidate the employers carry more weight).
%   - Experience before Education.
%
% Deliberate deviations, both narrow:
%   - A thin rule under each section heading. Stanford says avoid lines "unless
%     they help readability"; a single hairline is the standard scanning aid in
%     nearly every professional resume and earns its place.
%   - Up to two pages is allowed by Stanford. generate.js still targets one,
%     because a one-page resume for two years of professional experience is the
%     stronger document — but the layout no longer fights to achieve it.
%
% ATS constraints that override aesthetics, each measured against pdftotext:
%   - cmap MUST precede fontenc. Without it pdflatex emits fi/fl as ligature
%     glyphs with no reverse mapping and "profile" extracts as "prole".
%   - Headings use \MakeUppercase, never \scshape: small caps renders the first
%     letter full size and the rest small, and extraction yields "E DUCATION".
%   - Single column; contact details in the body, never a header; standard
%     section names; no icons or graphics.
%
% Placeholders are substituted by generate.js. Content comes only from
% master-facts.json — nothing is authored here.

\documentclass[letterpaper,11pt]{article}

\usepackage[margin=0.75in]{geometry}

\usepackage{cmap}                  % MUST precede fontenc
\usepackage[T1]{fontenc}
\usepackage[utf8]{inputenc}
\usepackage{tgheros}               % Helvetica clone ~ Arial
\renewcommand{\familydefault}{\sfdefault}
\usepackage{tabularx}
\usepackage[hidelinks]{hyperref}
\input{glyphtounicode}
\pdfgentounicode=1

\setlength{\parindent}{0pt}
\pagestyle{empty}

% Let long URLs in the contact line break instead of running into the margin.
% Without this the last link overflowed the right margin by ~0.3in — the ink
% reached 0.19in from the page edge while every other element stopped correctly.
\usepackage{url}
\urlstyle{same}     % URLs in the body font, not typewriter
\sloppy
\setlength{\emergencystretch}{3em}

% Section heading: bold uppercase, hairline rule beneath.
\newcommand{\resumesection}[1]{%
  \vspace{9pt}%
  {\normalsize\bfseries\MakeUppercase{#1}}\\[-4pt]%
  % \par, not \\: a \\ after the rule leaves an empty line box when the next
  % thing is a list, so Projects and Skills sat visibly lower than Experience,
  % where an \entry fills that line. \par ends the paragraph without one.
  \rule{\textwidth}{0.5pt}\par\vspace{2pt}%
}

% Title flush left, location/dates flush right.
%
% tabularx with an X column, not tabular*: tabular* cannot wrap, so a long left
% cell (an institution plus a degree plus a GPA) pushed the right-hand dates
% clean off the page — ink reached the paper edge with zero right margin. X
% wraps the left cell and keeps the dates inside the text block.
\newcommand{\entry}[2]{%
  \begin{tabularx}{\textwidth}{@{}X@{\hspace{1em}}r@{}}
    #1 & #2 \\
  \end{tabularx}%
}

% Bullets. NOTE: \setlength inside \begin{itemize} is silently ignored — \list
% resets these as it opens — so they are passed as \list's second argument.
\newenvironment{resumebullets}{%
  \begin{list}{\textbullet}{%
    \setlength{\leftmargin}{1.2em}%
    \setlength{\labelsep}{0.5em}%
    \setlength{\labelwidth}{0.7em}%
    \setlength{\itemsep}{2pt}%
    \setlength{\parsep}{0pt}%
    \setlength{\topsep}{4pt}%
    \setlength{\partopsep}{0pt}%
    \setlength{\rightmargin}{0pt}%
  }%
}{\end{list}}

\begin{document}

%% NAME %%
{\LARGE\bfseries __NAME__}\\[4pt]
%% CONTACT — body text, never a header: parsers skip headers %%
__CONTACT__\\[2pt]

__SUMMARY__

__EXPERIENCE__

__PROJECTS__

__SKILLS__

__EDUCATION__

\end{document}
