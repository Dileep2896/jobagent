% Resume template — Charter, single column, ATS-safe.
%
% Ported from the pattern Dileep's other tooling uses: Charter serif, tight
% geometry, small-caps section headings with a full-width rule, and an \entry
% command putting the title flush left and location/dates flush right.
%
% Two deliberate deviations from that pattern:
%
%   1. enumitem and titlesec are NOT used — neither is installed here
%      (they live in texlive-latex-extra). The same output is produced with a
%      plain \list and a hand-rolled heading macro, so this compiles against
%      texlive-latex-recommended alone.
%
%   2. cmap is added, and MUST precede fontenc. MEASURED: without it pdflatex
%      renders fi/fl as ligature glyphs with no reverse mapping, so "profile"
%      extracts as "prole" and "workflows" as "workows" — silently corrupting
%      the exact keywords an ATS matches on. The pdftotext gate in generate.js
%      fails the build if this regresses.
%
% The \entry rows use tabular*. VERIFIED against pdftotext: a one-row tabular
% for a title/date line extracts cleanly and in reading order. The standard ATS
% warning about tables concerns tables holding page structure, not this.
%
% Other ATS constraints held throughout: single column; contact details in the
% document body, never a header (parsers skip headers); standard section names;
% no icons or graphics.
%
% Placeholders are substituted by generate.js. Content comes only from
% master-facts.json — nothing is authored here.

\documentclass[letterpaper,10pt]{article}

\usepackage[left=0.5in,right=0.5in,top=0.45in,bottom=0.45in]{geometry}

\usepackage{cmap}                  % MUST precede fontenc — see note 2 above
\usepackage[T1]{fontenc}
\usepackage{charter}               % the professional serif look
\usepackage[utf8]{inputenc}
\usepackage{tabularx}
\usepackage[hidelinks]{hyperref}   % links work but render black, print clean
\input{glyphtounicode}
\pdfgentounicode=1

\setlength{\parindent}{0pt}
\pagestyle{empty}

% Section heading: bold UPPERCASE with a full-width rule beneath.
%
% The source pattern used \scshape here. MEASURED and rejected: small caps
% renders the first letter full-size and the rest small, and pdftotext reads
% the result as "E DUCATION" — a parser looking for "Education" does not match.
% \MakeUppercase carries the same visual weight and extracts as "EDUCATION".
\newcommand{\resumesection}[1]{%
  \vspace{4pt}%
  {\normalsize\bfseries\MakeUppercase{#1}}\\[-5.5pt]%
  \rule{\textwidth}{0.6pt}\\[2pt]%
}

% Title flush left, location/dates flush right, on one line.
\newcommand{\entry}[2]{%
  \begin{tabular*}{\textwidth}{@{}l@{\extracolsep{\fill}}r@{}}
    #1 & #2 \\
  \end{tabular*}%
}

% Tight bullets without enumitem. NOTE: \setlength inside \begin{itemize} is
% silently ignored — \list resets these as it opens — so they are passed as
% \list's second argument instead.
\newenvironment{resumebullets}{%
  \begin{list}{\textbullet}{%
    \setlength{\leftmargin}{1.35em}%
    \setlength{\labelsep}{0.5em}%
    \setlength{\labelwidth}{0.85em}%
    \setlength{\itemsep}{1pt}%
    \setlength{\parsep}{0pt}%
    \setlength{\topsep}{1pt}%
    \setlength{\partopsep}{0pt}%
    \setlength{\rightmargin}{0pt}%
  }%
}{\end{list}}

\begin{document}

%% NAME %%
{\LARGE\bfseries __NAME__}\\[2pt]
%% CONTACT — body text, never a header: parsers skip headers %%
__CONTACT__\\[1pt]

__SUMMARY__

__EXPERIENCE__

__PROJECTS__

__SKILLS__

__EDUCATION__

\end{document}
