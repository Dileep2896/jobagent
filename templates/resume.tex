%==============================================================
% RESUME TEMPLATE — Dileep Kumar Sharma
% ATS-friendly, single-column, one page.
%
% This is the user's proven template. generate.js substitutes the __PLACEHOLDER__
% tokens; everything else — fonts, margins, spacing, section order — is theirs and
% is not to be "improved" without measuring first. Two earlier rewrites of this
% file lost ground by following published style guides over what actually renders.
%
% HAND USE
%   Fill every __PLACEHOLDER__, delete unused blocks, then:  pdflatex Resume.tex
%
% PAGE FILL — the gate that matters
%   gs -q -dNOPAUSE -dBATCH -sDEVICE=bbox Resume.pdf 2>&1 | grep HiResBoundingBox
%   Letter is 792pt tall. With a 0.45in bottom margin (32.4pt), content's bottom-y
%   should land within ~10pt of it. Above ~60pt the page is under-filled: add a
%   bullet, a project, or a coursework line. generate.js enforces this in code
%   (SLACK_TARGET_PT / SLACK_UNDERFILL_PT) and grows the content to reach it.
%   Measuring ink HEIGHT instead of this bottom gap is what let a visibly thin
%   page pass as "93% full".
%
% LaTeX GOTCHAS (generate.js escapes these; they matter for hand use)
%   %  -> \%      &  -> \&      $  -> \$      |  -> \textbar{}
%   em dash -> ---            literal ~ -> \textasciitilde
%
% SPACING KNOBS (to fill or shrink the page)
%   geometry margins ....... 0.4in (tight) to 0.6in (loose)
%   itemsep / topsep ....... 1pt (tight) to 2.5pt (loose)
%   \titlespacing .......... {before}{after}, e.g. {4pt}{2pt} tight, {6pt}{3pt} loose
%==============================================================
\documentclass[letterpaper,10pt]{article}

\usepackage[left=0.5in,right=0.5in,top=0.45in,bottom=0.45in]{geometry}

% cmap before fontenc so pdflatex emits reverse glyph mappings. Charter extracts
% fi/fl correctly without it, verified with pdftotext — this is free insurance, not
% a fix for an observed bug.
\usepackage{cmap}
\usepackage{charter}                       % clean professional serif = the whole "look"
\usepackage[T1]{fontenc}
\usepackage{enumitem}
\usepackage{titlesec}
\usepackage{xcolor}
\usepackage{tabularx}
\usepackage[hidelinks]{hyperref}           % links work but render BLACK (ATS-safe)
\input{glyphtounicode}
\pdfgentounicode=1

\setlength{\parindent}{0pt}
\pagestyle{empty}

% Bullet spacing (main knob for page fill)
\setlist[itemize]{leftmargin=1.35em, itemsep=1.5pt, topsep=1.5pt,
                  parsep=0pt, partopsep=0pt, label=\textbullet}

% Section heading: the small-caps LOOK without \scshape.
%
% Real small caps renders the initial full size and the rest from the SC font, and
% pdftotext reads the font change as a word break: "S UMMARY", "E XPERIENCE",
% "PATENTS & P UBLICATIONS". Section headings are exactly what an ATS uses to
% segment a resume, so that is a real cost for a cosmetic feature.
%
% Instead: real uppercase characters throughout, first letter at full size and the
% remainder at \small. Visually near-identical to \scshape; extracts as clean
% "SUMMARY" / "EXPERIENCE" / "PATENTS & PUBLICATIONS".
%
% Two arguments because splitting the string in TeX is fragile — xstring chokes on
% the \& in "Patents \& Publications". Arg 1 is the first letter, arg 2 the rest,
% both already uppercase:  \resumesection{S}{UMMARY}
% titlesec setup is verbatim from the original template except for the dropped
% \scshape; only the text handed to \section changes.
\titleformat{\section}
  {\normalsize\bfseries}{}{0em}{}[\vspace{-5.5pt}\rule{\textwidth}{0.6pt}]
\titlespacing{\section}{0pt}{4pt}{2pt}
\newcommand{\resumesection}[2]{\section*{#1{\small #2}}}

% \entry{bold title on the left}{date/location flush right}
%
% tabular* with an l column cannot wrap, so an over-long left cell silently pushes
% the right cell off the page. At 10pt over a 7.5in text block that has room to
% spare — the earlier failure came from setting 11pt on a 6.5in block, not from
% this pattern. runGates() fails the build on any Overfull \hbox, so a regression
% here is caught rather than shipped.
\newcommand{\entry}[2]{%
  \begin{tabular*}{\textwidth}{@{}l@{\extracolsep{\fill}}r@{}}
    \textbf{#1} & #2 \\
  \end{tabular*}\par\vspace{1pt}}
